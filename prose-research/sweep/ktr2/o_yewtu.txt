    <script>
      // Self-host the cap WASM (no external CDN).
      //
      // ?v=2 — ONE-TIME cache poisoning escape, not a version knob.
      // Cache-busting is automatic: the app serves an ETag + Cache-Control:
      // no-cache, so browsers revalidate and pick up changes on their own, and
      // that keeps working — this parameter never needs bumping for a normal
      // deploy. It exists because of a past incident: nginx briefly served the
      // OLD bundle at these stable URLs still carrying the previous
      // `Cache-Control: public, max-age=31536000, immutable` header. Every
      // visitor in that window pinned that copy in their BROWSER cache until
      // 2027, and `immutable` means they never revalidate — so they kept running
      // a stale bundle no matter what the server sends (measured: fresh page
      // loads still POSTing the long-removed "client_timeout"). Server-side
      // purging cannot reach those entries; changing the URL is the only escape.
      window.CAP_CUSTOM_WASM_URL =
        "/antibot/assets/cap_wasm_bg.wasm?v=2";
    </script>
    <script>
      // Resilient image-captcha fallback (plain ES5 — MUST parse on very old
      // engines: PaleMoon/Goanna, legacy Firefox, etc.). The challenge scripts
      // below use modern JS (optional chaining, WASM, WebCrypto); a browser that
      // can't PARSE them just aborts that <script>, <noscript> does NOT fire
      // (JS is enabled), and the JS-driven fallback never appears — the visitor
      // is stuck on the spinner. This watchdog runs FIRST and reveals the image
      // captcha if the challenge errors out or never starts.
      (function () {
        var revealed = false, sent = {};
        // Best-effort telemetry so unsupported browsers are visible in the app's
        // logging backend. Uses an <img> GET (works on the oldest engines). One
        // beacon per distinct reason — a stray runtime error must not consume the
        // slot a genuine parse_error would need. Logged via /antibot/api/client-report.
        function report(reason, e) {
          if (sent[reason]) return;
          sent[reason] = true;
          try {
            var q = "reason=" + reason + "&t=" + (+new Date());
            // Without the message we cannot tell a SyntaxError in our own bundle
            // from an extension throwing on an unrelated script — which is what
            // made this signal unreadable before.
            if (e) {
              if (e.message) q += "&msg=" + encodeURIComponent(String(e.message).slice(0, 200));
              if (e.filename) q += "&file=" + encodeURIComponent(String(e.filename).slice(0, 200));
              if (e.lineno) q += "&line=" + encodeURIComponent(String(e.lineno));
            }
            new Image().src = "/antibot/api/client-report?" + q;
          } catch (err) { /* best effort */ }
        }
        function reveal(msg) {
          if (revealed) return;
          revealed = true;
          var fb = document.getElementById("antibot-fallback");
          if (fb) fb.hidden = false;
          var link = document.getElementById("antibot-captcha-link");
          if (link) {
            link.href = "/antibot/captcha?redirect=" +
              encodeURIComponent(location.pathname + location.search);
          }
          // href="" resolves to the current document, so the link reloads even
          // on engines too old to have run the module bundle. Setting it
          // explicitly keeps the query string and drops any fragment.
          var rl = document.getElementById("antibot-reload");
          if (rl) rl.href = location.pathname + location.search;
          var info = document.getElementById("antibot-info");
          if (info) info.style.display = "none";
          var st = document.getElementById("antibot-status");
          if (st && msg) st.textContent = msg;
        }
        window.__antibotReveal = reveal;
        // Is this error from a script WE serve, or from someone else's?
        // Same-origin scripts report a real filename; extension content scripts
        // report moz-extension://… / chrome-extension://…, and cross-origin ones
        // report "" with the message "Script error.".
        function isOurScript(f) {
          return typeof f === "string" && f.indexOf("/antibot/assets/") !== -1;
        }
        // Registered before the challenge scripts load, so it catches their
        // parse/runtime errors. e.message is present for script errors and
        // absent for resource-load failures, so unrelated 404s don't trigger it.
        //
        // Only OUR scripts failing is fatal. This used to reveal the captcha on
        // any uncaught error, which was wrong: window.onerror is global, so an
        // extension or an unrelated inline script throwing anywhere on the page
        // tripped it. Measured over 24h, 63 of 93 visitors reported "parse_error"
        // and then passed the JS challenge normally — they were shown "your
        // browser couldn't run the automatic check" while it was, in fact,
        // running. Errors from elsewhere are still reported (as script_error) so
        // they stay visible, but they no longer derail a working challenge.
        // A bundle that genuinely cannot parse is still covered twice: its own
        // filename lands here, and the not_started backstop below catches the rest.
        window.addEventListener("error", function (e) {
          if (!e || !e.message) return;
          var fatal = isOurScript(e.filename);
          report(fatal ? "parse_error" : "script_error", e);
          if (fatal) {
            reveal("Your browser couldn't run the automatic check — please solve an image challenge instead.");
          }
        }, true);
        // Backstop: the challenge module sets window.__antibotStarted the moment
        // it runs. If that hasn't happened a few seconds after load, the bundle
        // never parsed/started — offer the fallback.
        setTimeout(function () {
          if (!window.__antibotStarted) {
            report("not_started");
            reveal("This is taking longer than expected — you can solve an image challenge instead.");
          }
        }, 8000);
      })();
    </script>

      // Enrich the fallback link with the original destination for the JS path,
      // and populate the reference ID from the __antibot_ref cookie once it's set.
      (function () {
        var f = document.getElementById("antibot-captcha-link");
        if (f) f.href = "/antibot/captcha?redirect=" + encodeURIComponent(location.pathname + location.search);
        var rl = document.getElementById("antibot-reload");
        if (rl) rl.href = location.pathname + location.search;

        function updateRef() {
          var m = document.cookie.match(/(?:^|;\s*)__antibot_ref=([0-9a-f]+)/);
          var el = document.getElementById("antibot-ref-id");
          if (m && el && el.textContent !== m[1]) el.textContent = m[1];
        }
        updateRef();
        setInterval(updateRef, 500);
      })();
