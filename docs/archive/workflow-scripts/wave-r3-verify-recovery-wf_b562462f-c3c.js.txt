async function opusVerify(prompt, label) {
  let v = await agent(prompt, { label, phase: 'Verify', model: 'opus', effort: 'high', schema: VERIFY_SCHEMA })
  if (!v) {
    log(label + ' died (likely API overload) — retrying once')
    v = await agent(prompt, { label: label + ':retry', phase: 'Verify', model: 'opus', effort: 'high', schema: VERIFY_SCHEMA })
  }
  return v
}
