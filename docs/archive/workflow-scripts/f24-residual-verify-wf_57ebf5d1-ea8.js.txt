phase('Sweep')
const results = await parallel([
  ...HUNTERS.map(h => () => agent(h.prompt, { label: `hunt:${h.key}`, phase: 'Sweep', schema: FINDINGS })),
  ...REVIEWERS.map(r => () => agent(r.prompt, { label: `review:${r.key}`, phase: 'Sweep', schema: REVIEW, effort: r.effort })),
])
const hunterOut = results.slice(0, HUNTERS.length).filter(Boolean)
const reviewOut = results.slice(HUNTERS.length).filter(Boolean)

const raw = hunterOut.flatMap(r => r.findings)
const seen = new Set()
const deduped = raw.filter(f => { const k = `${f.file}:${f.line}`; if (seen.has(k)) return false; seen.add(k); return true })
log(`hunters: ${raw.length} raw findings, ${deduped.length} deduped; reviewers: ${reviewOut.flatMap(r => r.issues).length} issues`)

phase('Verify')
const verified = await parallel(deduped.map(f => () =>
  agent(`${COMMON}

A hunter claims this is surviving F24-class corruption. Try to REFUTE it — default to isReal=false if uncertain. Read the file and its full surrounding context, trace the data producer if the claim involves parsing, and check git history if intent is unclear.

Claim: ${f.file}:${f.line} — ${f.what}
Rationale: ${f.why_suspect}

isReal=true ONLY if you are confident this is genuine corruption (placeholder/artifact/mismatch) still present in the working tree that should be fixed.`, { label: `verify:${f.file.split('/').pop()}:${f.line}`, phase: 'Verify', schema: VERDICT })
    .then(v => ({ ...f, verdict: v }))
))

const confirmed = verified.filter(Boolean).filter(x => x.verdict.isReal)
return {
  confirmedResiduals: confirmed,
  refuted: verified.filter(Boolean).filter(x => !x.verdict.isReal).map(x => ({ file: x.file, line: x.line, what: x.what, why: x.verdict.reasoning })),
  reviewIssues: reviewOut.flatMap(r => r.issues),
}
