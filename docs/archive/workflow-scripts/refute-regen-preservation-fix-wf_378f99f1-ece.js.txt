phase('Refute')

const rounds = await pipeline(
  LENSES,
  lens => agent(`${PREAMBLE}\n\n${lens.prompt}\n\nReturn every defect you can substantiate. An empty findings array is acceptable ONLY if you genuinely could not substantiate anything after a serious attempt.`, {
    label: `refute:${lens.key}`,
    phase: 'Refute',
    schema: FINDINGS,
    effort: 'high',
  }),
  (result, lens) => {
    const found = (result && result.findings) || []
    if (found.length === 0) return []
    return parallel(found.map(f => () =>
      agent(`${PREAMBLE}\n\nA reviewer working the ${lens.key} lens reported this defect. Your job is to REFUTE it — assume it is wrong until the code forces you to agree. Read the actual code and, where possible, RUN something that settles it empirically (a scratch vitest file under /private/tmp is fine; do not add files to the repo's tests/ directory).\n\nTITLE: ${f.title}\nFILE: ${f.file}${f.line ? ':' + f.line : ''}\nCLAIM: ${f.claim}\nFAILURE SCENARIO: ${f.failure_scenario}\nEVIDENCE OFFERED: ${f.evidence}\n\nSet real=false if the claim is wrong, already handled elsewhere, cosmetic, or depends on conditions that cannot occur. Set real=true ONLY if you can show the defect actually bites. If real, give the minimal correct fix.`, {
        label: `verify:${lens.key}:${(f.title || '').slice(0, 34)}`,
        phase: 'Verify',
        schema: VERDICT,
        effort: 'high',
      }).then(v => ({ lens: lens.key, finding: f, verdict: v }))
    ))
  },
)

const judged = rounds.flat().filter(Boolean)
const confirmed = judged.filter(j => j.verdict && j.verdict.real)
const rejected = judged.filter(j => j.verdict && !j.verdict.real)

log(`refuted-and-survived: ${confirmed.length} of ${judged.length} raised`)

phase('Synthesize')

if (confirmed.length === 0) {
  return {
    confirmed: [],
    rejectedCount: rejected.length,
    rejectedTitles: rejected.map(r => r.finding.title),
    summary: 'No finding survived adversarial verification.',
  }
}

const synthesis = await agent(
  `${PREAMBLE}\n\nThe following defects SURVIVED independent adversarial verification against the reroll-preservation fix:\n\n${confirmed.map((c, i) => `${i + 1}. [${c.finding.severity}] (${c.lens}) ${c.finding.title}\n   file: ${c.finding.file}${c.finding.line ? ':' + c.finding.line : ''}\n   claim: ${c.finding.claim}\n   scenario: ${c.finding.failure_scenario}\n   verifier: ${c.verdict.reasoning}\n   suggested fix: ${c.verdict.correction || 'none given'}`).join('\n\n')}\n\nAlso rejected during verification (do not resurrect these): ${rejected.map(r => r.finding.title).join('; ') || 'none'}\n\nProduce a decision-ready synthesis for the engineer who wrote the fix:\n- Which findings are genuinely must-fix-before-this-ships, and which are follow-ups or owner decisions?\n- Are any of them duplicates of each other, or symptoms of one root cause?\n- For the must-fix set, give the concrete minimal change, in order.\n- Explicitly state whether the CORE claim still holds: does an authored NPC survive regenSection('npcs'), and is same-seed generation unaffected?\nBe concise and concrete. Prose, not bullets-of-bullets.`,
  { label: 'synthesize', phase: 'Synthesize', effort: 'high' },
)

return {
  confirmed: confirmed.map(c => ({
    lens: c.lens,
    severity: c.finding.severity,
    title: c.finding.title,
    file: c.finding.file,
    line: c.finding.line,
    claim: c.finding.claim,
    scenario: c.finding.failure_scenario,
    confidence: c.verdict.confidence,
    correction: c.verdict.correction,
  })),
  rejectedCount: rejected.length,
  rejectedTitles: rejected.map(r => r.finding.title),
  synthesis,
}
