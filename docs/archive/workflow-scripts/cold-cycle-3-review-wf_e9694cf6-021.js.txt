Work ONLY from the code in ${WORKTREE} (READ-ONLY — never edit, never run git mutations, never commit). const LENSES = [
  ['correctness', 'Hunt logic bugs in the domain engine: off-by-ones, inverted conditions, wrong aggregations, unreachable branches, contradictory derivations between modules. Read the deepest kernels (src/domain/worldPulse, src/domain/state, generators) with a skeptic eye.'],

const FINDINGS = {
  type: 'object', additionalProperties: false, required: ['findings'],
  properties: { findings: { type: 'array', maxItems: 12, items: {
    type: 'object', additionalProperties: false,
    required: ['file', 'line', 'severity', 'claim', 'failureScenario'],
    properties: {
      file: { type: 'string' }, line: { type: 'integer' },
      severity: { enum: ['C', 'H', 'M'] },
      claim: { type: 'string' }, failureScenario: { type: 'string' },
      probe: { type: 'string' },
    } } } },
}

const VERDICT = {
  type: 'object', additionalProperties: false, required: ['refuted', 'reason'],
  properties: { refuted: { type: 'boolean' }, reason: { type: 'string' } },
}

phase('Cold Find')
const rounds = await parallel(LENSES.map(([key, charge]) => () =>
  agent(`${COLD}\n\nYOUR LENS: ${charge}\n\nReturn your findings as structured output.`,
    { label: `find:${key}`, phase: 'Cold Find', schema: FINDINGS, model: 'opus' })
))

const all = rounds.filter(Boolean).flatMap((r, i) =>
  (r.findings || []).map(f => ({ ...f, lens: LENSES[i][0] })))
const seen = new Map()
for (const f of all) {
  const k = `${f.file}:${f.line}:${(f.claim || '').slice(0, 60).toLowerCase()}`
  if (!seen.has(k)) seen.set(k, f)
}
const deduped = [...seen.values()]
log(`${all.length} raw findings -> ${deduped.length} after dedup`)

phase('Verify')
const REFUTE_LENSES = ['reproduce it at the code (does the scenario actually occur?)', 'check spec-consistency (is this intended behavior a cold reader misread?)', 'check blast-radius honesty (is the severity inflated?)']
const verified = await parallel(deduped.map(f => () => {
  const panels = f.severity === 'M' ? REFUTE_LENSES.slice(0, 1) : REFUTE_LENSES
  return parallel(panels.map(lens => () =>
    agent(`${COLD}\n\nYou are an adversarial VERIFIER. Try hard to REFUTE this claimed finding via this lens: ${lens}\n\nFINDING: [${f.severity}] ${f.file}:${f.line} — ${f.claim}\nSCENARIO: ${f.failureScenario}\n\nDefault to refuted=true if you cannot positively confirm the defect at the code.`,
      { label: `verify:${f.file.split('/').pop()}`, phase: 'Verify', schema: VERDICT, model: 'opus' })
  )).then(votes => {
    const v = votes.filter(Boolean)
    const refutes = v.filter(x => x.refuted).length
    const survives = f.severity === 'M' ? refutes === 0 : refutes < 2
    return { ...f, survives, votes: v.map(x => ({ refuted: x.refuted, reason: x.reason.slice(0, 200) })) }
  })
}))

const confirmed = verified.filter(Boolean).filter(f => f.survives)
const killed = verified.filter(Boolean).filter(f => !f.survives)
log(`${confirmed.length} confirmed, ${killed.length} refuted`)
return {
  confirmed: confirmed.sort((a, b) => 'CHM'.indexOf(a.severity) - 'CHM'.indexOf(b.severity)),
  refuted: killed.map(f => ({ file: f.file, line: f.line, claim: f.claim, severity: f.severity, votes: f.votes })),
  stats: { raw: all.length, deduped: deduped.length, confirmed: confirmed.length },
}
