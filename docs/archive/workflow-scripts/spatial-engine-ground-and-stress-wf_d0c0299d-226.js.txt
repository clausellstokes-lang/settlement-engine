const ground = await parallel(GROUND_AREAS.map(a => () =>
  agent(`${preamble}\n\nYOUR SYSTEM: ${a.label}.\n${a.focus}\n\nReturn the structured facts. Be concrete with file paths and line ranges; mark feasibility honestly.`,
