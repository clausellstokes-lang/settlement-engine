// Parse the raw JSON strings defensively (strip any accidental fence).
const parsed = raw.map((s, i) => {
  if (!s || typeof s !== 'string') return { _error: 'empty', idx: i }
  let t = s.trim()
  const a = t.indexOf('{'); const b = t.lastIndexOf('}')
  if (a >= 0 && b > a) t = t.slice(a, b + 1)
  try { return JSON.parse(t) } catch (e) { return { _error: String(e).slice(0, 120), idx: i, head: s.slice(0, 200) } }
})

const ok = parsed.filter((p) => p && !p._error)
const bad = parsed.filter((p) => p && p._error)

return { reruns: parsed }
