export const meta = {
  name: 'pricing-compendium-design-research',
  description: 'Second design-research sweep: pricing-page + reference/codex design feeding THE DOCUMENTATION WAVE',
  phases: [{ title: 'Research' }, { title: 'Critique' }],
}
const FINDINGS = {
  type: 'object',
  properties: {
    findings: { type: 'array', items: { type: 'object', properties: {
      source: { type: 'string' }, url: { type: 'string' },
      pattern: { type: 'string' }, emulate: { type: 'string' }, avoid: { type: 'string' },
    }, required: ['source', 'pattern', 'emulate'] } },
    synthesis: { type: 'string' },
  },
  required: ['findings', 'synthesis'],
}
phase('Research')
const ANGLES = [
  { key: 'pricing-design', prompt: `Research the best-designed pricing pages on the web — tier-card layout, visual hierarchy, the one-primary-CTA discipline, honest-pricing exemplars (no dark patterns), how the best pages make a 3-4 tier ladder legible in seconds, and how they present what each tier UNLOCKS in mechanism terms rather than adjective lists. Return concrete layout/hierarchy patterns with emulate/avoid.` },
  { key: 'hobby-market-pricing', prompt: `Research how games/hobby/TTRPG products present pricing and what THIS audience trusts or distrusts: Foundry VTT (famous one-time-purchase stance), World Anvil's tier ladder, Inkarnate, D&D Beyond, Roll20, Dungeon Alchemist (one-time on Steam), plus community sentiment (Reddit r/rpg, r/DMToolkit threads about subscription fatigue vs one-time purchases, what pricing moves got tools praised or roasted).   { key: 'credit-economy-presentation', prompt: `Research how AI-era products present usage-based/credit pricing HONESTLY and digestibly: per-task price menus (image generators, AI writing tools, API pricing pages), how the best ones answer "what will this actually cost me in practice", cost estimators/calculators, labeled-estimates language, and BYOK (bring-your-own-key) framing where users pay a platform fee + their own inference. Find pages praised for clarity and pages criticized for token-opacity. The client's law: "users buy outcomes, not tokens" — task-priced credits (e.g. interpret-session 5cr, construct-realm 8cr) with packs at volume discounts, estimates labeled estimates, and a usage meter. Return concrete presentation patterns (menus, calculators, worked examples) with emulate/avoid.` },
  { key: 'reference-codex-design', prompt: `Research the best reference/encyclopedia/codex interface design for the client's Compendium page: developer docs done beautifully (Stripe docs, MDN), in-game codexes and encyclopedias famous for making enumerations delightful (Civilization's Civilopedia, Elden Ring/Tunic-style lore books, Dwarf Fortress legends viewers, well-designed game wikis), searchable catalog UIs, and long-enumeration layout patterns (40+ archetype cards, 300-institution catalogs) that stay browsable rather than overwhelming. const results = await parallel(ANGLES.map(a => () =>
  agent(a.prompt + ' Use WebSearch and WebFetch tools for real, current sources. Cite URLs.', { label: `research:${a.key}`, phase: 'Research', schema: FINDINGS })))
phase('Critique')
const critic = await agent(`You are the completeness critic for the SECOND design-research sweep feeding SettlementForge's DOCUMENTATION WAVE (three pages, one argument: About=why trust, Compendium=proof, Pricing=convert; parchment/gold house style; subscription-fatigued TTRPG audience; four product goals: best map generator / settlement generator / campaign simulator / TTRPG AI). return { angles: ANGLES.map((a, i) => ({ key: a.key, result: results[i] })), critic }
