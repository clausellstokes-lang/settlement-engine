phase('Find')
const results = await pipeline(
  DIMENSIONS,
  (d) => agent(
    `You are an adversarial code reviewer. Review ONLY the M11b diff (\`git show ${COMMIT}\`, and read the enclosing functions + the full calamity.js/calamityKernel.js as needed).\n${CONTEXT}\n\nYOUR DIMENSION — ${d.key}: ${d.focus}\n\nReturn up to 6 candidate findings, each with a CONCRETE failure scenario (specific inputs/state → wrong output / crash / byte-drift / conservation-break / invariant-break / budget-growth). Only findings with a nameable failure scenario. If clean, return empty. No pure style.`,
    { label: `find:${d.key}`, phase: 'Find', schema: FINDINGS_SCHEMA, model: 'opus' },
  ),
  (review, d) => parallel((review?.findings || []).map((f) => () =>
    parallel([0, 1, 2].map((i) => () =>
      agent(
        `You are refuter #${i + 1} (of 3), adversarially trying to REFUTE a code-review finding on the M11b calamity mover. Default to refuted=true if not constructible from the actual code.\n${CONTEXT}\n\nTHE FINDING (dimension ${d.key}):\nfile: ${f.file}:${f.line}\nsummary: ${f.summary}\nfailure_scenario: ${f.failure_scenario}\n\nRead the actual code (git show ${COMMIT} + the file) and decide. REFUTED only if: factually wrong (quote the line), provably impossible (cite type/constant/guard/invariant), already handled (cite the guard), or pure style. Otherwise CONFIRMED (constructible now) or PLAUSIBLE (realistic rare path). refuted=true for REFUTED, false otherwise.`,
        { label: `verify:${d.key}:${f.line}:r${i + 1}`, phase: 'Verify', schema: VERDICT_SCHEMA, model: 'opus' },
      )
    )).then((votes) => {
      const v = votes.filter(Boolean)
      return { ...f, dimension: d.key, votes: v, refutedCount: v.filter((x) => x.refuted).length, survives: v.filter((x) => x.refuted).length < 2 }
    })
  )),
)

const all = results.flat().filter(Boolean)
const survivors = all.filter((f) => f.survives)
const rank = { high: 0, med: 1, low: 2 }
survivors.sort((a, b) => (rank[a.severity] ?? 3) - (rank[b.severity] ?? 3))
log(`M11b review: ${all.length} candidates → ${survivors.length} survived 3-refuter verify`)
return {
  totalCandidates: all.length, survived: survivors.length,
  findings: survivors.map((f) => ({
    severity: f.severity, dimension: f.dimension, file: f.file, line: f.line,
    summary: f.summary, failure_scenario: f.failure_scenario, refutedCount: f.refutedCount,
  })),
}
