const results = await pipeline(
  DIMENSIONS,
  d => agent(d.prompt, { label: `review:${d.key}`, phase: 'Review', schema: FINDINGS }),
  (review, d) => {
    const found = (review?.findings || []).filter(f => f && f.title)
    if (!found.length) return []
    return parallel(found.map(f => () =>
      agent(`${COMMON}
A reviewer claims this defect in the W-Session wave. Adversarially VERIFY it — your default stance is that the reviewer is wrong; read the actual code (and the consumed modules' real shapes) and only confirm if the concrete failure scenario truly occurs. For Foundry-API claims, reason carefully about Foundry v11-v13 documented behavior and be explicit about uncertainty.
CLAIM (dimension ${d.key}): ${JSON.stringify(f)}`,
        { label: `verify:${(f.file || '').split('/').pop()}:${f.title.slice(0, 30)}`, phase: 'Verify', schema: VERDICT })
        .then(v => ({ ...f, dimension: d.key, verdict: v }))
    ))
  },
)

const all = results.filter(Boolean).flat().filter(Boolean)
const confirmed = all.filter(f => f.verdict?.isReal && f.verdict.severityConfirmed !== 'not-a-bug')
const rejected = all.filter(f => !(f.verdict?.isReal) || f.verdict?.severityConfirmed === 'not-a-bug')
log(`confirmed ${confirmed.length} / rejected ${rejected.length} of ${all.length} findings`)
return { confirmed, rejected: rejected.map(f => ({ title: f.title, file: f.file, reason: f.verdict?.reasoning?.slice(0, 200) })) }
