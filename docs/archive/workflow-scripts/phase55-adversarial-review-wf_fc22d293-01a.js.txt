function diffHeader(c) {
  return `COMMIT ${c.sha} — ${c.title} (mover-ladder step ${c.code}).
The checkout at ${REPO} is already on the branch under review (review-fixes-2026-07-08 HEAD).
Get the diff: \`cd ${REPO} && git show ${c.sha} --stat\` then for each SOURCE file \`git diff ${c.sha}^..${c.sha} -- <path>\`.
Review SOURCE files ONLY (src/** or lib/**). SKIP *.test.* / *.spec.* / *.md / *.json / *.sql. Use Read for surrounding context and Grep for callers when tracing cross-file impact. Line numbers must refer to the CURRENT file content (Read to confirm), not diff offsets.`
}

function correctnessPrompt(c) {
  return `You are an adversarial CORRECTNESS reviewer auditing ONE commit of SettlementForge, a deterministic JS D&D world-simulator. READ-ONLY: report defects, do NOT fix.

${diffHeader(c)}
${INVARIANTS}
YOUR ANGLES (correctness + determinism only — the other reviewer handles cleanup):
(A) LINE-BY-LINE diff scan: inverted/negated conditions, off-by-one, null/undefined deref, missing await on a promise, falsy-zero bugs (0 or '' treated as "absent"), copy-paste errors (wrong var/index reused), swallowed errors (empty catch / ignored rejection), NON-DETERMINISM (Date.now/new Date/Math.random, or iteration over an unsorted Map/Set/object that feeds output), unrounded float persisted to saved state.
(B) REMOVED-BEHAVIOR audit: for every DELETED or replaced line in the diff, is its invariant re-established elsewhere? Pay special attention to any removed guard, gate, sort, rounding, or dormancy/opt-in check — did the removal silently break byte-identity or dormancy?
(C) CROSS-FILE trace: for each changed/added exported function, Grep its callers across ${REPO}/src. Does a new precondition, a changed return shape, a newly-thrown exception, or a changed argument order break an existing call site?

Be exhaustive but precise: report only defects with a CONCRETE, constructible failure scenario. Prefer fewer, real findings over speculation. ${c.crit ? 'This commit touches constitutional invariants — scrutinize determinism, dormancy gating, and the frozen digest especially hard.' : ''}
Return findings via the structured schema. If the commit is clean on these angles, return an empty findings array.`
}

function cleanupPrompt(c) {
  return `You are an adversarial CLEANUP + ALTITUDE reviewer auditing ONE commit of SettlementForge, a deterministic JS D&D world-simulator. READ-ONLY: report, do NOT fix.

${diffHeader(c)}
${INVARIANTS}
YOUR ANGLES (cleanup, altitude, conventions — the other reviewer handles correctness):
(D) REUSE: does the new code re-implement a helper that already exists? Grep ${REPO}/src for an existing function that does the same thing (distance reads, sorting, rounding, PRNG forking, digest access, clamp/lerp, etc.).
(E) SIMPLIFICATION: redundant or derivable state (a field recomputed from others), dead code (unreachable branch, unused export/param/var), needless indirection.
(F) EFFICIENCY: redundant recompute on the per-tick or per-route HOT PATH (e.g. rebuilding a map/sort inside a loop that could hoist out), a closure capturing a large scope it doesn't need, an O(N^2) where O(N log N) is trivial and N is settlements/edges.
(G) ALTITUDE: a bandaid where the real fix belongs deeper; a special-case branch layered onto shared infra that will need re-patching for the next case; a frozen-digest overlay that should be read-time but instead special-cases the capture.
(H) CONVENTIONS: violates a constitutional invariant above, or a clear local idiom in the surrounding file (quote the idiom/line it breaks).

Report only findings with concrete justification. A cleanup finding must name what to reuse/remove/hoist and why it's safe. ${c.crit ? 'Watch for altitude problems around the frozen digest and dormancy gating especially.' : ''}
Return findings via the structured schema. If clean, return an empty findings array.`
}

function verifyPrompt(c, f) {
  return `Adversarially VERIFY ONE code-review finding against SettlementForge, a deterministic JS world-sim. The checkout at ${REPO} is already on the branch under review. Your DEFAULT posture is to REFUTE — try to prove the finding wrong.

FINDING (commit ${c.sha}, ladder ${c.code}):
  location: ${f.file}:${f.line}
  angle/category: ${f.angle}/${f.category}   claimed severity: ${f.severity}
  claim: ${f.summary}
  failure scenario: ${f.failure_scenario}

Inspect the ACTUAL code: \`git show ${c.sha} -- ${f.file}\` for the change, Read ${REPO}/${f.file} for current context, Grep for the guards/callers that would prevent or trigger the scenario.
${INVARIANTS}
VERDICT rules (recall-biased — keep real defects, kill fake ones):
- REFUTED: the defect is constructibly wrong, impossible given the types/domain, or ALREADY GUARDED by a check/round/sort/gate elsewhere. You MUST cite the exact guard at file:line.
- CONFIRMED: you traced the concrete logic path that yields the wrong output / crash / broken invariant; state the exact trigger.
- PLAUSIBLE: a realistic world-state reaches the code and you cannot fully confirm or refute without executing — default HERE when genuinely unsure (do not refute on a hunch).
Set invariant_risk to the constitutional invariant threatened (byte-identity / dormancy / frozen-digest / codepoint-sort / budget / master-merge), or 'none'. Do NOT downgrade a genuine invariant violation. Give final severity (a broken invariant is 'high').
Return the structured verdict with a one-paragraph justification citing exact file:line evidence.`
}

phase('Review')
log(`Reviewing ${COMMITS.length} Phase 5.5 source commits — two find-lenses each, then adversarial verify.`)

const perCommit = await pipeline(
  COMMITS,
  // stage 1: two find-lenses in parallel; merge + light-dedup findings
  (c) => parallel([
    () => agent(correctnessPrompt(c), { schema: FIND_SCHEMA, phase: 'Review', label: `find:${c.code}:corr` }),
    () => agent(cleanupPrompt(c),     { schema: FIND_SCHEMA, phase: 'Review', label: `find:${c.code}:clean` }),
  ]).then((rs) => {
    const all = rs.filter(Boolean).flatMap((r) => (r && r.findings) ? r.findings : [])
    const seen = new Set()
    const findings = []
    for (const f of all) {
      const k = `${f.file}::${f.line}::${f.category}`
      if (seen.has(k)) continue
      seen.add(k)
      findings.push(f)
    }
    return { c, findings }
  }),
  // stage 2: adversarial verify each finding (runs as soon as this commit's find completes)
  (merged) => parallel(
    merged.findings.map((f) => () =>
      agent(verifyPrompt(merged.c, f), { schema: VERDICT_SCHEMA, phase: 'Verify', label: `verify:${merged.c.code}:${(f.file || '').split('/').pop()}:${f.line}` })
        .then((v) => ({ commit: merged.c.sha, code: merged.c.code, title: merged.c.title, ...f, verdict: v }))
        .catch(() => null)
    )
  ).then((arr) => ({ code: merged.c.code, sha: merged.c.sha, title: merged.c.title, verified: arr.filter(Boolean) }))
)

const commits = perCommit.filter(Boolean)
const allVerified = commits.flatMap((r) => r.verified || [])
const surviving = allVerified.filter((f) => f.verdict && f.verdict.verdict !== 'REFUTED')
const confirmed = surviving.filter((f) => f.verdict.verdict === 'CONFIRMED')
const plausible = surviving.filter((f) => f.verdict.verdict === 'PLAUSIBLE')
const refuted = allVerified.filter((f) => f.verdict && f.verdict.verdict === 'REFUTED')
const invariantHits = surviving.filter((f) => f.verdict.invariant_risk && f.verdict.invariant_risk !== 'none')

log(`Done. ${allVerified.length} candidates verified: ${confirmed.length} CONFIRMED, ${plausible.length} PLAUSIBLE, ${refuted.length} REFUTED. ${invariantHits.length} touch a constitutional invariant.`)

const perCommitSummary = commits.map((r) => ({
  code: r.code, sha: r.sha, title: r.title,
  confirmed: (r.verified || []).filter((f) => f.verdict && f.verdict.verdict === 'CONFIRMED').length,
  plausible: (r.verified || []).filter((f) => f.verdict && f.verdict.verdict === 'PLAUSIBLE').length,
  refuted: (r.verified || []).filter((f) => f.verdict && f.verdict.verdict === 'REFUTED').length,
}))

return {
  totals: { candidates: allVerified.length, confirmed: confirmed.length, plausible: plausible.length, refuted: refuted.length, invariantHits: invariantHits.length },
  perCommitSummary,
  invariantHits: invariantHits.map((f) => ({ code: f.code, file: f.file, line: f.line, summary: f.summary, verdict: f.verdict.verdict, risk: f.verdict.invariant_risk, severity: f.verdict.severity })),
  surviving: surviving.map((f) => ({
    code: f.code, sha: f.commit, file: f.file, line: f.line, angle: f.angle, category: f.category,
    verdict: f.verdict.verdict, severity: f.verdict.severity, invariant_risk: f.verdict.invariant_risk,
    summary: f.summary, failure_scenario: f.failure_scenario, justification: f.verdict.justification,
  })),
}
