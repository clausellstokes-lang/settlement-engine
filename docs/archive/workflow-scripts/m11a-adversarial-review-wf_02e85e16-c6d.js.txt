const FINDINGS_SCHEMA = {
  type: 'object', additionalProperties: false,
  properties: {
    findings: {
      type: 'array',
      items: {
        type: 'object', additionalProperties: false,
        properties: {
          file: { type: 'string' },
          line: { type: 'number' },
          summary: { type: 'string' },
          failure_scenario: { type: 'string' },
          severity: { type: 'string', enum: ['high', 'med', 'low'] },
        },
        required: ['file', 'line', 'summary', 'failure_scenario', 'severity'],
      },
    },
  },
  required: ['findings'],
}

const VERDICT_SCHEMA = {
  type: 'object', additionalProperties: false,
  properties: {
    refuted: { type: 'boolean' },
    verdict: { type: 'string', enum: ['CONFIRMED', 'PLAUSIBLE', 'REFUTED'] },

phase('Find')
const results = await pipeline(
  DIMENSIONS,
  (d) => agent(
    `You are an adversarial code reviewer. Review ONLY the M11a diff (run \`git diff ${RANGE}\` and read the enclosing functions of every hunk; also read the whole of pestilence.js + pestilenceKernel.js as needed).\n${CONTEXT}\n\nYOUR DIMENSION — ${d.key}: ${d.focus}\n\nReturn up to 6 candidate findings, each with a CONCRETE failure scenario (specific inputs/state → wrong output / crash / byte-drift / budget-growth / invariant-break). Only findings with a nameable failure scenario. If the dimension is clean, return an empty findings array. Do NOT flag pure style.`,
    { label: `find:${d.key}`, phase: 'Find', schema: FINDINGS_SCHEMA, model: 'opus' },
  ),
  (review, d) => parallel((review?.findings || []).map((f) => () =>
    parallel([0, 1, 2].map((i) => () =>
      agent(
        `You are refuter #${i + 1} (of 3), adversarially trying to REFUTE a code-review finding on the M11a pestilence mover. Default to refuted=true if the finding is not constructible from the actual code.\n${CONTEXT}\n\nTHE FINDING (dimension ${d.key}):\nfile: ${f.file}:${f.line}\nsummary: ${f.summary}\nfailure_scenario: ${f.failure_scenario}\n\nRead the actual code (git diff ${RANGE} + the file) and decide. REFUTED only if: factually wrong (quote the line), provably impossible (cite the type/constant/guard/invariant), already handled in the diff (cite the guard), or pure style with no observable effect. Otherwise CONFIRMED (constructible now) or PLAUSIBLE (realistic rare path). Set refuted=true for REFUTED, false for CONFIRMED/PLAUSIBLE.`,
        { label: `verify:${d.key}:${f.line}:r${i + 1}`, phase: 'Verify', schema: VERDICT_SCHEMA, model: 'opus' },
      )
    )).then((votes) => {
      const v = votes.filter(Boolean)
      const refutes = v.filter((x) => x.refuted).length
      return { ...f, dimension: d.key, votes: v, refutedCount: refutes, survives: refutes < 2 }
    })
  )),
)

const all = results.flat().filter(Boolean)
const survivors = all.filter((f) => f.survives)
const rank = { high: 0, med: 1, low: 2 }
survivors.sort((a, b) => (rank[a.severity] ?? 3) - (rank[b.severity] ?? 3))
log(`M11a review: ${all.length} candidates → ${survivors.length} survived 3-refuter verify`)
return {
  totalCandidates: all.length,
  survived: survivors.length,
  findings: survivors.map((f) => ({
    severity: f.severity, dimension: f.dimension, file: f.file, line: f.line,
    summary: f.summary, failure_scenario: f.failure_scenario, refutedCount: f.refutedCount,
  })),
}
