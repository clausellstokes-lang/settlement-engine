phase('Census')
const findings = await pipeline(
  SURFACES,
  (s) => agent(`${PREAMBLE}\n\n${s.prompt}`, { label: `census:${s.key}`, phase: 'Census', schema: SCHEMA }),
  (result, s) => {
    if (!result) return null
    const claims = (result.impacts || []).filter((i) => i.severity !== 'benign')
    if (!claims.length) return { surface: s.key, result, verdicts: [] }
    return parallel(claims.slice(0, 6).map((c) => () =>
      agent(`${PREAMBLE}

ADVERSARIAL VERIFICATION. Another agent claims the following about surface "${s.key}":

  FILE: ${c.file}:${c.line}
  CLAIM: ${c.what}
  SEVERITY: ${c.severity}
  THEIR EVIDENCE: ${c.evidence}

Your job is to REFUTE this claim. Open the cited file and read the actual code. Check whether
the quoted evidence exists verbatim and whether it actually supports the claim. Common ways such
claims are wrong: the cited line does not exist or says something else; a default/fallback arm
already handles the case; the code path is gated off and never runs; the claim confuses the
audit receipt sink with the canonical feed; the claim assumes a field is required when the
consumer is absent-tolerant.
Default to refuted=true when you cannot positively confirm the claim from source you read.`,
        { label: `verify:${c.file.split('/').pop()}:${c.line}`, phase: 'Verify', schema: {
          type: 'object',
          additionalProperties: false,
          required: ['refuted', 'reasoning', 'correctedClaim'],
          properties: {
            refuted: { type: 'boolean' },
            reasoning: { type: 'string' },
            correctedClaim: { type: 'string', description: 'If the claim was partly right, the accurate version; else empty' },
          },
        } })
        .then((v) => ({ claim: c, verdict: v }))
    )).then((verdicts) => ({ surface: s.key, result, verdicts: verdicts.filter(Boolean) }))
  },
)

const live = findings.filter(Boolean)
log(`censused ${live.length}/${SURFACES.length} surfaces`)

phase('Synthesize')
const confirmed = live.flatMap((f) =>
  (f.verdicts || []).filter((v) => v.verdict && !v.verdict.refuted)
    .map((v) => ({ surface: f.surface, ...v.claim, corrected: v.verdict.correctedClaim })))

const summary = await agent(`${PREAMBLE}

You are synthesizing a consumer census. Below are the per-surface findings and the
adversarially-CONFIRMED impacts (claims that survived a refutation attempt).

RAW FINDINGS:
${JSON.stringify(live.map((f) => ({ surface: f.surface, impacts: f.result.impacts, breakageRisks: f.result.breakageRisks, notes: f.result.notes })), null, 2)}

CONFIRMED AFTER REFUTATION ATTEMPT:
${JSON.stringify(confirmed, null, 2)}

Produce the implementer's disposition ledger. For each downstream surface state plainly:
what breaks, what changes behavior, what is benign. Then give a RANKED list of the concrete
edits the fix must include beyond the two author files, with file:line. Call out explicitly:
 - the exact id shape to use for each of the six kinds, and whether satelliteId (or any other
   component) is REQUIRED to avoid a same-tick id collision that would silently merge beats
 - every exact-ceiling line-count test that the fix must stay under, with current/ceiling numbers
 - every test/pin that must be updated in the same commit
Be concrete and terse. Report uncertainty explicitly rather than guessing.`,
  { label: 'synthesize', phase: 'Synthesize' })

return { summary, confirmed, surfaces: live.map((f) => ({ surface: f.surface, notes: f.result.notes })) }
