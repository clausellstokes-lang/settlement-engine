phase('Audit')
const audited = await pipeline(
  LENSES,
  (lens) => agent(lens.prompt, { label: `audit:${lens.key}`, phase: 'Audit', schema: FINDINGS }),
  (res, lens) => {
    if (!res || !res.findings || res.findings.length === 0) return { lens: lens.key, findings: [], verdicts: [] }
    return agent(`${RULES}

You are an ADVERSARIAL VERIFIER. Another agent produced these findings about the
site-derivation surface. Your job is to REFUTE each one. Default to real=false when
uncertain. A finding is NOT real if: it is INTENDED design (siteGenesis explicitly says
substance is derived from "resources that imply landforms" — an export implying a landform
is the DESIGN, so only a genuinely incoherent or false-positive-driven result is a defect);
or the code path is unreachable; or the claim misreads the control flow; or the evidence is
inspection-only and inspection is wrong. Verify by READING THE ACTUAL CODE and, where
useful, by running a node probe under /tmp/sca-probes/. Judge each on its merits.

FINDINGS TO REFUTE:
${JSON.stringify(res.findings, null, 2)}`, { label: `verify:${lens.key}`, phase: 'Confirm', schema: VERDICTS })
      .then((v) => ({ lens: lens.key, findings: res.findings, verdicts: (v && v.verdicts) || [], notes: res.notes }))
  },
)

const rows = audited.filter(Boolean)
const survived = []
for (const r of rows) {
  const byTitle = new Map((r.verdicts || []).map((v) => [v.title, v]))
  for (const f of (r.findings || [])) {
    const v = byTitle.get(f.title)
    if (v && v.real === false) continue
    survived.push({ ...f, lens: r.lens, severity: (v && v.correctedSeverity) || f.severity })
  }
}
log(`audit complete: ${rows.length} lenses, ${survived.length} findings survived refutation`)
