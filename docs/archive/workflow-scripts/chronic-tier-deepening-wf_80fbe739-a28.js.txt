export const meta = {
  name: 'chronic-tier-deepening',
  description: 'Deepen chronic-kind pools to the frequency-scaled floor + author the legacy annex',
  phases: [

    { title: 'Verify', detail: '3 conformance checkers' },

  legacyTokens:{type:'array',items:{type:'object',required:['kind','currentVariants','cadence'],properties:{
    kind:{type:'string'},currentVariants:{type:'number'},cadence:{type:'string'}}}}}}

  annex:{type:'string'},kindsDeepened:{type:'number'},variantsAdded:{type:'number'},samples:{type:'array',items:{type:'string'}}}}
const VSCHEMA = { type:'object', required:['scope','violationsFixed','belowFloor','verdict'], properties:{
  scope:{type:'string'},violationsFixed:{type:'number'},belowFloor:{type:'array',items:{type:'string'}},verdict:{type:'string'}}}

phase('Census')

  }),
  () => agent(`${PRE}\n${VOICE_PTR}\nASSIGNMENT (legacy author): create ${MF}/docs/content/RECEIPT_POOLS_LEGACY.md for the LIVE routed tokens the census found below floor: ${JSON.stringify((census.legacyTokens||[]).slice(0,200))}. Header: note these pools RETROFIT live kinds — wiring is a DISCLOSED same-seed prose shift under the spine's frequency-scaled-floor amendment (the recorded ruling; per-kind golden plan lands at wiring; variant 1 of every pool MUST be the kind's CURRENT live string VERBATIM so the dark path stays byte-identical and the pool only ever widens). Return {annex:'LEGACY', kindsDeepened, variantsAdded, samples: 5}.`, {label:'author:legacy', phase:'Author', model:'opus', schema:WSCHEMA}),
])

phase('Verify')

  () => agent(`${PRE}\nVERIFIER (adversarial; fix in place): re-check ${MF}/docs/content/RECEIPT_POOLS_{WAR,TRADE,FAITH}.md deepened blocks against the annex-header contract + the frequency floor: angle-distinctness (collapse slot-swaps and re-author), digits, slots, belief attribution, Law One, no renumbering of pre-existing variants (verify against git diff).   () => agent(`${PRE}\nVERIFIER (adversarial; fix in place): same check for ${MF}/docs/content/RECEIPT_POOLS_{POPULATIONS,INFORMATION,GRAMMAR,INTERIOR,COUPLINGS}.md. Return {scope:'LEGACY', violationsFixed, belowFloor, verdict}.`, {label:'verify:legacy', phase:'Verify', model:'opus', schema:VSCHEMA}),
])
return { census: {chronic: (census.perAnnexChronic||[]).length, legacy: (census.legacyTokens||[]).length}, authored: results.filter(Boolean), verified: ver.filter(Boolean) }
