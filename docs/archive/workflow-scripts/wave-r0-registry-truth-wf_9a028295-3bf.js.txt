const FIX_SCHEMA = {
  type: 'object',
  required: ['filesTouched', 'pins', 'testsRun', 'testOutputTail', 'behaviorShift', 'deferrals'],
  properties: {
    filesTouched: { type: 'array', items: { type: 'string' } },
    pins: { type: 'number' },
    testsRun: { type: 'string' },
    testOutputTail: { type: 'string', description: 'verbatim final lines of each test run' },
    behaviorShift: { type: 'string', description: 'exactly what user-visible behavior changed, or "none"' },
    deferrals: { type: 'array', items: { type: 'string' } },
    notes: { type: 'string' },
    error: { type: 'string' },
  },
}

const VERIFY_SCHEMA = {
  type: 'object',
  required: ['verdict', 'mustFix', 'checked'],
  properties: {
    verdict: { type: 'string', enum: ['PASS', 'PASS_WITH_NOTES', 'MUST_FIX'] },
    mustFix: { type: 'array', items: { type: 'string' } },
    checked: { type: 'array', items: { type: 'string' } },
    notes: { type: 'string' },
  },
}

function fixPrompt(l) {
  return COMMON + '\n\nYOUR LANE OWNS ONLY THESE FILES: ' + l.files + '\n\n' + l.brief + '\n\nDeliver: working code + pins + verbatim test output. Deferral over scope-creep, always.'
}

function verifyPrompt(l, r) {
  return COMMON + `\n\nYOU ARE THE OPUS ADVERSARIAL VERIFIER for lane ${l.key}. The fixer reports: files=${JSON.stringify(r.filesTouched)}; pins=${r.pins}; behaviorShift=${r.behaviorShift}; tests="${r.testsRun}".\nREPO IS READ-ONLY FOR YOU except you MAY run tests. Protocol:
1. Read every touched file's full diff context (git diff -- <file> shows combined foreign+lane changes — attribute hunks carefully; the fixer's changes are the ones matching the lane brief).
2. Try to REFUTE: does the fix hold on every lifecycle path (create/read/persist/regen/undo/restore)? Does any same-seed output move? Any new eager import (check the import graph)? Any capped file grown past its exact ceiling? Any hand-rolled faction/tier access where a cure function exists? Tooltip/copy honest against the COMMITTED lineage, not the uncommitted fix?
3. Re-run the fixer's tests yourself single-threaded PLUS one suite the fixer did NOT run that the touched files plausibly affect.
4. For lane registry-truth specifically: re-derive 10 of the null-row classifications independently; refute any 'armed' decision whose undo does not actually restore (run the pin).
5. For lane deity-gate specifically: verify the second mount path end-to-end and that free-tier refusal cannot be reached from a state where the UI promised availability.
Verdict MUST_FIX blocks the wave — list each item concretely. PASS_WITH_NOTES for real-but-deferable observations.`
}
