phase('Review')

const results = await pipeline(
  DIMENSIONS,
  d => agent(`${PREAMBLE}\nDIMENSION: ${d.key}\n${d.prompt}`, { label: `find:${d.key}`, phase: 'Review', schema: FIND_SCHEMA }),
  (review, d) => {
    if (!review || !review.findings || review.findings.length === 0) return { dimension: d.key, review, verify: null }
    const findingsText = review.findings.map((f, i) => `[${i + 1}] ${f.title} (${f.severity}, ${f.confidence}) — ${f.file}\n${f.detail}\nFailure: ${f.failureScenario}`).join('\n\n')
    return agent(`${PREAMBLE}\nYou are the ADVERSARIAL VERIFIER for review dimension "${d.key}". A reviewer reported the findings below. Your job is to REFUTE each one: re-derive it from the live tree, execute the cheapest decisive check (read the file, run the focused test, run the grep), and only let it survive if the evidence is real. Findings that are true-but-intentional (documented deferrals, design-doc-sanctioned behavior, recorded hazards already guarded) get REFUTED with the citation. Default to REFUTED when the evidence is ambiguous. Severity-correct anything that survives.\n\nFINDINGS:\n${findingsText}`, { label: `verify:${d.key}`, phase: 'Verify', model: 'opus', schema: VERIFY_SCHEMA })
      .then(v => ({ dimension: d.key, review, verify: v }))
  }
)

phase('Critic')
const summary = results.filter(Boolean).map(r => {
  const kept = r.verify ? r.verify.verdicts.filter(v => v.verdict !== 'REFUTED').length : 0
  const total = r.review && r.review.findings ? r.review.findings.length : 0
  return `${r.dimension}: ${total} found, ${kept} survived verification`
}).join('\n')

const critic = await agent(`${PREAMBLE}\nYou are the COMPLETENESS CRITIC for a 10-dimension holistic review of the Opus-era SettlementForge work (dimensions: ${DIMENSIONS.map(d => d.key).join(', ')}; results summary:\n${summary}\n). Your question: WHAT DID THE REVIEW MISS? Consider: surfaces no dimension owned (edge functions/supabase, PDF export of new content, a11y of new tabs, the soak/certification harness itself, migration train 189-193 coherence, security of new DM verbs — can a free-tier user reach them?, analytics of new events); cross-cutting risks no single-file review sees (two new systems each fine alone but composing badly — demographics×war×routes feedback loops); and lifecycle paths (regen/undo/import of every new ledger). Spend your time on the 3-5 highest-consequence blind spots, verify each is REAL with a quick look at the tree, and return them as findings.`, { label: 'completeness-critic', phase: 'Critic', schema: FIND_SCHEMA })

return { dimensions: results.filter(Boolean), critic }
