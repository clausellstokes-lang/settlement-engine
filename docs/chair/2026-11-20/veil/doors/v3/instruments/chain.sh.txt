#!/bin/sh
# chain.sh <db> <list: N | a+b+c where a may be a ceiling "203"> <migdir>  (REVIEW-VEIL-11's, S= re-pointed; scaffold = the doors' walker scaffold)
set -u
DB=$1; LIST=$2; MIG=$3
S=/var/tmp/vd3
export PGHOST=/var/tmp/vd3/sock PGUSER=postgres
psql -qd postgres -c "drop database if exists $DB with (force)" -c "create database $DB" >/dev/null
psql -qd $DB -v ON_ERROR_STOP=1 -f $S/nat/scaffold.sql >/dev/null 2>$S/nat/scaffold-err.txt || echo "scaffold failed: $(head -c 300 $S/nat/scaffold-err.txt)"
apply() { sed -E 's/create extension if not exists (pg_cron|pg_net)( with schema extensions)?/select 1/gI' "$1" > $S/nat/mig.$DB.sql
  psql -qd $DB -1 -v ON_ERROR_STOP=1 -f $S/nat/mig.$DB.sql >/dev/null 2>$S/nat/err.$DB.txt || echo "FAIL $(basename $1): $(head -c 300 $S/nat/err.$DB.txt)"; }
CEIL=$(echo $LIST | cut -d+ -f1); EXTRA=$(echo $LIST | cut -s -d+ -f2-)
for f in $(ls $MIG | grep -E '^[0-9]{3}_.*\.sql$' | sort); do
  n=$(expr $(echo $f | cut -c1-3) + 0)
  if [ $n -le $CEIL ] || echo "+$EXTRA+" | grep -q "+$n+"; then apply $MIG/$f; fi
done
psql -qd $DB -c "insert into auth.users(id,email) values ('11111111-1111-1111-1111-111111111111','o@x.test'),('22222222-2222-2222-2222-222222222222','r@x.test') on conflict do nothing; insert into public.profiles (id) values ('11111111-1111-1111-1111-111111111111'),('22222222-2222-2222-2222-222222222222') on conflict do nothing; update public.profiles set role = 'developer' where id = '11111111-1111-1111-1111-111111111111'" >/dev/null
echo "built $DB ($LIST)"
