async function year(saves, worldState, y) {
  const { fullRules } = composeSoakRules({ preset: SIMULATION_RULE_PRESETS.full_simulation.rules, seasons: 'preset', overlay: {} });
  const campaign = { id: 'wf0-live-year', name: 'WF-0 live year', settlementIds: saves.map((s) => s.id), regionalGraph: ensureRegionalGraph({ edges: [], channels: [] }, { now: NOW }),
    wizardNews: { currentTick: 0, entries: [] },
    worldState: worldState || { rngSeed: SEED, tick: 0, canonizedAt: NOW, ...buildWholeWorldSoakSpatialCanon(saves), simulationRules: fullRules, stressors: [] } };
  const result = await simulateCampaignWorldInterval({ campaign, saves, interval: 'one_year', commit: true, now: NOW, autoResolve: true, advanceEpoch: soakAdvanceEpoch({ simulationRules: fullRules, seed: SEED, year: y }) });
  const byId = new Map((result.settlementUpdates || []).map((u) => [String(u.saveId), u.settlement]));
  return { worldState: result.worldState, finalSaves: saves.map((s) => (byId.has(String(s.id)) ? { ...s, settlement: byId.get(String(s.id)) } : s)) };
}
const brief = (recs) => recs.map((r) => `${r.name} [${r.coreMotif?.element}/${r.coreMotif?.act} band ${r.scaleBand}]${drawsPilgrims(r) ? ' *PILGRIMS*' : ''}`);
test('WF0-MEASURE', () => {
  const generated = realm(WF0_DEITY_BEARING_CASE.customContent);
  const { saves } = seatDeityBearingCase(generated, WF0_DEITY_BEARING_CASE, { now: NOW });
  const digest = buildWholeWorldSoakSpatialCanon(saves).spatialDigest; const out = {};
  for (const id of SEQ_IDS) { const seated = saves.find((s) => s.id === id); const raw = generated.find((s) => s.id === id);
    const q = deriveFoundingTraditions(seated.settlement).filter(drawsPilgrims); const reach = saves.filter((s) => s.id !== id && isMapped(digest, s.id) && pathCost(digest, id, s.id) != null).map((s) => s.id);
    out[id] = { name: seated.name, patron: Boolean(seated.settlement?.config?.primaryDeitySnapshot), cults: (seated.settlement?.config?.cultDeitySnapshots || []).length,
      seatedSet: brief(deriveFoundingTraditions(seated.settlement)), unseatedSet: brief(deriveFoundingTraditions(raw.settlement)), seatedQualifying: q.length,
      unseatedQualifying: deriveFoundingTraditions(raw.settlement).filter(drawsPilgrims).length, mapped: isMapped(digest, id), reachable: reach,
      draw: q.length ? pilgrimageDraw({ hostId: id, hostRec: q[0], settlements: saves.map((s) => ({ id: s.id, settlement: s.settlement })), digest }) : null };
    for (const host of ['soak-a', 'soak-b']) out[`plan:${host}`] = planDeityBearingSeats(generated, { ...WF0_DEITY_BEARING_CASE, patronHostId: host }, { now: NOW }).preconditionsMet; }
  writeFileSync(process.env.WF0_OUT, JSON.stringify(out, null, 1));
}, 600000);
