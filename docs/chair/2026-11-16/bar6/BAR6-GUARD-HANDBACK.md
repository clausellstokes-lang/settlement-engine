`printed.mjs <hits.json> <render-dir> <out.json>` (hits from `scanTree`):

```js
// which hits print in a PAGE-COPY render: the longest fixed fragment (18+ chars) of the line found in the dumps
import fs from 'node:fs'; import path from 'node:path';
const [hitsPath, renderDir, outPath] = process.argv.slice(2);
const hits = JSON.parse(fs.readFileSync(hitsPath, 'utf8'));
const dumps = fs.readdirSync(path.join(renderDir, 'dumps')).filter((x) => x.endsWith('.txt'));
const pages = dumps.map((x) => fs.readFileSync(path.join(renderDir, 'dumps', x), 'utf8'));
for (const h of hits) {
  const frags = h.text.split(/\$\{[^}]*\}|\{[^}]*\}|…|\\n|\n/).map((s) => s.trim()).sort((a, b) => b.length - a.length);
  const fr = frags[0] || ''; if (fr.length < 18) { h.printed = null; continue; }
  h.printed = pages.reduce((n, p) => n + (p.includes(fr) ? 1 : 0), 0);
}
fs.writeFileSync(outPath, JSON.stringify(hits));
const files = {}; for (const h of hits) { const f = files[h.file] ||= { n: 0, p: 0 }; f.n++; if (h.printed) f.p++; }
const dirs = {}; for (const [f, v] of Object.entries(files)) { const d = f.split('/').slice(0, 3).join('/'); const x = dirs[d] ||= { files: 0, pfiles: 0, n: 0, p: 0 }; x.files++; x.n += v.n; x.p += v.p; if (v.p) x.pfiles++; }
console.log('hits', hits.length, 'printed', hits.filter((h) => h.printed).length, 'unmeasurable', hits.filter((h) => h.printed === null).length, 'files', Object.keys(files).length, 'printing files', Object.values(files).filter((v) => v.p).length);
for (const [d, x] of Object.entries(dirs).sort((a, b) => b[1].n - a[1].n)) console.log(`${String(x.n).padStart(4)} hits ${String(x.p).padStart(4)} printed · files ${x.files}/${x.pfiles} printing · ${d}`);
```
