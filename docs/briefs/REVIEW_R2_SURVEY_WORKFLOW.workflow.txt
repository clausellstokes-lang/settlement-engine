function subBrief(s) {
  return PREAMBLE + '\n\nYOUR SLICE: ' + s.key + '\nSCOPE: ' + s.scope + '\nSLICE-SPECIFIC ATTENTION: ' + s.notes + '\n\nSet slice="' + s.key + '" in your output. Prefix finding ids with "' + s.key + '".'
}
function dimBrief(d) {
  return PREAMBLE + '\n\nYOU ARE A CROSS-CUTTING DIMENSION REVIEWER (not a file-scope reader): sample the whole tree via targeted greps + full reads of the load-bearing files for your dimension.\nDIMENSION: ' + d.key + '\nCHARGE: ' + d.charge + '\nSTART POINTERS: ' + d.pointers + '\n\nSet slice="dim:' + d.key + '" in your output. Prefix finding ids with "' + d.key + '".'
}

log('Dispatching ' + SUBSYSTEMS.length + ' subsystem readers + ' + DIMENSIONS.length + ' dimension reviewers')

const thunks = [
  ...SUBSYSTEMS.map(s => () => agent(subBrief(s), { label: 'read:' + s.key, phase: 'Subsystems', schema: SCHEMA })),
  ...DIMENSIONS.map(d => () => agent(dimBrief(d), { label: 'dim:' + d.key, phase: 'Dimensions', schema: SCHEMA })),
]

const results = await parallel(thunks)

const subs = results.slice(0, SUBSYSTEMS.length).filter(Boolean)
const dims = results.slice(SUBSYSTEMS.length).filter(Boolean)
const missing = []
results.forEach((r, i) => {
  if (!r) missing.push(i < SUBSYSTEMS.length ? SUBSYSTEMS[i].key : 'dim:' + DIMENSIONS[i - SUBSYSTEMS.length].key)
})
const totalFindings = [...subs, ...dims].reduce((n, r) => n + (r.findings ? r.findings.length : 0), 0)
log('Landed ' + subs.length + '/' + SUBSYSTEMS.length + ' subsystems, ' + dims.length + '/' + DIMENSIONS.length + ' dimensions, ' + totalFindings + ' findings. Missing: ' + (missing.join(', ') || 'none'))

return { subsystems: subs, dimensions: dims, missingSlices: missing, totalFindings }
