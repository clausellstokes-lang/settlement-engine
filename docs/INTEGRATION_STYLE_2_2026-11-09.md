```js
// usage: node fielddiff.mjs <treeA> <treeB> [--rows N]  — diffs every golden-corpus settlement leaf by path.
// Prints per top-level field: rows moved; and every NON-STRING leaf change (type or value) with examples.
import { pathToFileURL } from 'node:url';
import { resolve } from 'node:path';
const [a, b] = process.argv.slice(2).map((p) => resolve(p));
const load = async (root) => ({
  gen: (await import(pathToFileURL(`${root}/src/generators/generateSettlementPipeline.js`).href)).generateSettlementPipeline,
  corp: await import(pathToFileURL(`${root}/tests/helpers/goldenMasterCorpus.js`).href),

const flat = (v, p, out) => {
  if (v && typeof v === 'object') { const ks = Object.keys(v); if (!ks.length) out.set(p, Array.isArray(v) ? '[]' : '{}'); for (const k of ks) flat(v[k], `${p}.${k}`, out); }
  else out.set(p, v);
  return out;
};
const top = {}; const nonString = {}; let rowsMoved = 0; const fieldSet = new Set();

  const sa = A.gen(cfg, null, { seed: _seed, customContent: {} });
  const sb = B.gen(cfg, null, { seed: _seed, customContent: {} });
  if (JSON.stringify(sa) === JSON.stringify(sb)) continue;

  const fa = flat(sa, '', new Map()); const fb = flat(sb, '', new Map());
  const keys = new Set([...fa.keys(), ...fb.keys()]);
  const tops = new Set();
  for (const k of keys) {
    const x = fa.get(k), y = fb.get(k);
    if (x === y) continue;
    const t = k.split('.')[1]; tops.add(t);
    const norm = k.replace(/\.\d+(?=\.|$)/g, '.#');
    fieldSet.add(norm);
    if (!(typeof x === 'string' && typeof y === 'string')) {
      const e = (nonString[norm] ??= { n: 0, ex: [] }); e.n++; if (e.ex.length < 3) e.ex.push([A.corp.keyOf(config), x, y]);
    }
  }
  for (const t of tops) top[t] = (top[t] ?? 0) + 1;
}
```
