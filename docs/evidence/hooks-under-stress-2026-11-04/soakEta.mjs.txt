const STRESS_KEYS = Object.keys(STRESS_TYPE_MAP);
const STRESS_CELLS = ['none', 'random', ...STRESS_KEYS, 'pair', 'pair', 'pair'];
const tierAtLeast = (tier, floor) => TIER_ORDER.indexOf(tier) >= TIER_ORDER.indexOf(floor);
const allows = (type, tier) => !STRESS_TYPE_MAP[type]?.requiresTier || tierAtLeast(tier, STRESS_TYPE_MAP[type].requiresTier);
const PAIRS = []; for (let a = 0; a < STRESS_KEYS.length; a += 1) for (let b = a + 1; b < STRESS_KEYS.length; b += 1) PAIRS.push([STRESS_KEYS[a], STRESS_KEYS[b]]);
function stressForIndex(i, tier) {
  const block = Math.floor(i / TIER_ORDER.length); const cellIndex = block % STRESS_CELLS.length; const cell = STRESS_CELLS[cellIndex];
  if (cell === 'none') return { cell, keys: { selectedStressesRandom: false, selectedStresses: [], stressType: null, stressTypes: [] } };
  if (cell === 'random') return { cell, keys: {} };
  if (cell === 'pair') {
    const pairCell = cellIndex - (STRESS_CELLS.length - 3);
    const ordinal = ((Math.floor(block / STRESS_CELLS.length) * 3 + pairCell) * TIER_ORDER.length + TIER_ORDER.indexOf(tier)) % PAIRS.length;
    for (let t = 0; t < PAIRS.length; t += 1) { const pair = PAIRS[(ordinal + t) % PAIRS.length]; if (pair.every((x) => allows(x, tier))) return { cell: 'pair', keys: { stressTypes: [...pair], stressType: null } }; }
  }
  return { cell, keys: { stressType: cell, stressTypes: [cell] } };
}

function eta(cat, num) {
  const groups = new Map(); cat.forEach((c, i) => { if (!groups.has(c)) groups.set(c, []); groups.get(c).push(num[i]); });
  const m = num.reduce((a, b) => a + b, 0) / num.length; const sst = num.reduce((a, x) => a + (x - m) ** 2, 0);
  let ssb = 0; for (const g of groups.values()) { const gm = g.reduce((a, b) => a + b, 0) / g.length; ssb += g.length * (gm - m) ** 2; }
  return sst ? Math.sqrt(ssb / sst) : 0;
}
